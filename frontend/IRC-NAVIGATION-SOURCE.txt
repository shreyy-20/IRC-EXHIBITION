import type { EventInfo, Exhibitor, Booth, RouteResult } from './types'

const API = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8000/api'

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(`${API}${path}`, {
    ...init,
    headers: { 'Content-Type': 'application/json', ...(init?.headers || {}) },
  })
  if (!response.ok) {
    const body = await response.json().catch(() => ({}))
    throw new Error(body.detail || `Request failed (${response.status})`)
  }
  return response.json() as Promise<T>
}

export const getEvent = (id: string) => request<EventInfo>(`/events/${id}`)
export const getExhibitors = (id: string, q = '') =>
  request<Exhibitor[]>(`/events/${id}/exhibitors${q ? `?q=${encodeURIComponent(q)}` : ''}`)
export const getBooths = (hallId: string) => request<Booth[]>(`/halls/${hallId}/booths`)
export const getRoute = (start_node: string, destination_node: string) =>
  request<RouteResult>('/navigation/route', {
    method: 'POST',
    body: JSON.stringify({ start_node, destination_node }),
  })

from datetime import datetime, timezone
from sqlalchemy import String, Text, Float, ForeignKey, Boolean, DateTime, Integer, UniqueConstraint
from sqlalchemy.orm import Mapped, mapped_column, relationship
from app.database import Base

def now_utc():
    return datetime.now(timezone.utc)

class Event(Base):
    __tablename__ = "events"
    id: Mapped[str] = mapped_column(String(64), primary_key=True)
    name: Mapped[str] = mapped_column(String(200), nullable=False)
    description: Mapped[str] = mapped_column(Text, default="")
    venue: Mapped[str] = mapped_column(String(200), default="")
    is_demo: Mapped[bool] = mapped_column(Boolean, default=True)
    is_published: Mapped[bool] = mapped_column(Boolean, default=True)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=now_utc)
    exhibitors: Mapped[list["Exhibitor"]] = relationship(back_populates="event", cascade="all, delete-orphan")
    halls: Mapped[list["Hall"]] = relationship(back_populates="event", cascade="all, delete-orphan")

class Hall(Base):
    __tablename__ = "halls"
    id: Mapped[str] = mapped_column(String(64), primary_key=True)
    event_id: Mapped[str] = mapped_column(ForeignKey("events.id"), nullable=False, index=True)
    name: Mapped[str] = mapped_column(String(100), nullable=False)
    map_key: Mapped[str] = mapped_column(String(100), default="sample-hall")
    event: Mapped[Event] = relationship(back_populates="halls")
    booths: Mapped[list["Booth"]] = relationship(back_populates="hall", cascade="all, delete-orphan")

class Exhibitor(Base):
    __tablename__ = "exhibitors"

def login(payload: LoginRequest, db: Session = Depends(get_db)):
    from app.models import AdminUser
    user = db.query(AdminUser).filter_by(username=payload.username, is_active=True).first()

@app.post("/api/navigation/route", response_model=RouteResponse)
def calculate_route(payload: RouteRequest, db: Session = Depends(get_db)):
    nodes = {n.id for n in db.scalars(select(NavigationNode)).all()}
    if payload.start_node not in nodes or payload.destination_node not in nodes:
        raise HTTPException(status_code=404, detail="Starting point or destination was not found")
    edges = db.scalars(select(NavigationEdge).where(NavigationEdge.is_blocked == False)).all()
    graph = {}
    for edge in edges:
        graph.setdefault(edge.from_node, []).append((edge.to_node, edge.distance_m))
        graph.setdefault(edge.to_node, []).append((edge.from_node, edge.distance_m))
    queue = [(0.0, payload.start_node, [payload.start_node])]
    best = {}
    while queue:
        distance, node, path = heapq.heappop(queue)
        if node in best and best[node] <= distance: continue
        best[node] = distance
        if node == payload.destination_node:
            return RouteResponse(nodes=path, distance_m=round(distance, 2), walking_minutes=round(distance / 75, 1),
                message="Route calculated from the configured demonstration navigation graph.")
        for neighbour, weight in graph.get(node, []):
            if neighbour not in best: heapq.heappush(queue, (distance + weight, neighbour, path + [neighbour]))
    raise HTTPException(status_code=404, detail="No walkable route exists between these locations")

@app.post("/api/qr/generate")
def generate_qr(payload: dict, db: Session = Depends(get_db)):
    event_id = str(payload.get("event_id", ""))
    destination_id = str(payload.get("destination_id", ""))
    label = str(payload.get("destination_label", destination_id))
    if not db.get(Event, event_id):
        raise HTTPException(status_code=404, detail="Event not found")
    record_id = uuid.uuid4().hex[:16]
    db.add(QRCodeRecord(id=record_id, event_id=event_id, destination_id=destination_id, destination_label=label))
    db.add(AnalyticsEvent(event_id=event_id, event_type="qr_generated", entity_id=destination_id))
    db.commit()
    url = f"{settings.allowed_origins[0]}/?event={event_id}&destination={destination_id}" if settings.allowed_origins else f"http://localhost:5173/?event={event_id}&destination={destination_id}"
    image = qrcode.make(url)
    buffer = io.BytesIO(); image.save(buffer, format="PNG")
    return Response(content=buffer.getvalue(), media_type="image/png", headers={"X-QR-ID": record_id})

@app.post("/api/admin/import-exhibitors")
async def import_exhibitors(file: UploadFile = File(...), event_id: str = Query(...), db: Session = Depends(get_db), admin=Depends(require_admin)):
    if not db.get(Event, event_id):
        raise HTTPException(status_code=404, detail="Event not found")
    if not file.filename or not file.filename.lower().endswith(".csv"):
        raise HTTPException(status_code=400, detail="Upload a CSV file")
    content = (await file.read()).decode("utf-8-sig")
    reader = csv.DictReader(io.StringIO(content))
    required = {"id", "name"}
    if not reader.fieldnames or not required.issubset(set(reader.fieldnames)):
        raise HTTPException(status_code=400, detail="CSV requires id and name columns")
    created = 0
    for row in reader:
        if not row.get("id") or not row.get("name"): continue
        if db.get(Exhibitor, row["id"]): continue
        db.add(Exhibitor(id=row["id"], event_id=event_id, name=row["name"],
            category=row.get("category") or "General", description=row.get("description") or "",
            website=row.get("website") or ""))
        created += 1
    db.commit()
    return {"created": created}

    db.add(edge); db.commit()

from sqlalchemy.orm import Session

from app.models import (
    Event,

    Exhibitor,

from app.security import hash_password


def seed_demo(db: Session) -> None:
    # 1. Create the demo event if it does not exist.
    event = db.get(Event, "demo-irc-expo")

    if not event:
        event = Event(
            id="demo-irc-expo",
            name="IRC Exhibition Finder - Sample Exhibition",
            description=(
                "Demonstration event with fictional exhibitors "
                "and an illustrative floor map."
            ),
            venue="Sample Exhibition Venue",
            is_demo=True,
        )
        db.add(event)
        db.commit()

    # 2. Create the hall independently of the event check.
    hall = db.get(Hall, "demo-hall-1")

    if not hall:
        hall = Hall(
            id="demo-hall-1",
            event_id=event.id,
            name="Sample Hall 1",
            map_key="sample-hall-1",
        )
        db.add(hall)
        db.commit()

    # 3. Create demo exhibitors and booths.
    samples = [
        ("ex-001", "RoadTech Systems", "Road Infrastructure",
         "Demonstration exhibitor for road construction technology.",
         "B01", 100, 120),

        ("ex-002", "BridgeWorks India", "Bridge Engineering",
         "Demonstration exhibitor for bridge engineering.",
         "B02", 270, 120),

        ("ex-003", "SafeRoute Technologies", "Road Safety",
         "Demonstration exhibitor for road safety solutions.",
         "B03", 440, 120),

        ("ex-004", "GreenPavement Labs", "Materials",
         "Demonstration exhibitor for sustainable pavement materials.",
         "B04", 100, 300),

        ("ex-005", "InfraData Analytics", "Technology",
         "Demonstration exhibitor for infrastructure data.",
         "B05", 270, 300),
    ]

    for eid, name, category, description, code, x, y in samples:
        exhibitor = db.get(Exhibitor, eid)

        if not exhibitor:
            exhibitor = Exhibitor(
                id=eid,
                event_id=event.id,
                name=name,
                category=category,
                description=description,
            )
            db.add(exhibitor)

        booth_id = f"booth-{code.lower()}"
        booth = db.get(Booth, booth_id)

        if not booth:
            db.add(
                Booth(
                    id=booth_id,
                    hall_id=hall.id,
                    exhibitor_id=eid,
                    code=code,
                    x=x,
                    y=y,
                )
            )

    # 4. Create navigation nodes.
    node_data = [
        ("entrance", "Main Entrance", "entrance", 45, 420),
        ("junction-a", "Main Junction", "junction", 350, 190),
        ("junction-b", "North Aisle", "junction", 350, 150),
        ("junction-c", "South Aisle", "junction", 350, 330),
    ]

    for node_id, label, node_type, x, y in node_data:
        if not db.get(NavigationNode, node_id):
            db.add(
                NavigationNode(
                    id=node_id,
                    hall_id=hall.id,
                    label=label,
                    node_type=node_type,
                    x=x,
                    y=y,
                )
            )

    for i in range(1, 6):
        node_id = f"booth-b{i:02}"

        if not db.get(NavigationNode, node_id):
            db.add(
                NavigationNode(
                    id=node_id,
                    hall_id=hall.id,
                    label=f"Booth B{i:02}",
                    node_type="booth",
                )
            )

    # Flush nodes before creating edges that refer to them.
    db.flush()

    # 5. Create navigation edges only after the hall and nodes exist.
    edges = [
        ("entrance", "junction-a", 12),
        ("junction-a", "junction-b", 18),
        ("junction-a", "junction-c", 20),
        ("junction-b", "booth-b01", 10),
        ("junction-b", "booth-b02", 14),
        ("junction-b", "booth-b03", 25),
        ("junction-c", "booth-b04", 12),
        ("junction-c", "booth-b05", 15),
    ]

    for from_node, to_node, distance in edges:
        existing_edge = (
            db.query(NavigationEdge)
            .filter_by(
                hall_id=hall.id,
                from_node=from_node,
                to_node=to_node,
            )
            .first()
        )

        if not existing_edge:
            db.add(
                NavigationEdge(
                    hall_id=hall.id,
                    from_node=from_node,
                    to_node=to_node,
                    distance_m=distance,
                )
            )

    # 6. Create the local demo administrator if missing.
    admin = (
        db.query(AdminUser)
        .filter_by(username="admin")
        .first()
    )

    if not admin:
        db.add(
            AdminUser(
                username="admin",
                password_hash=hash_password(
                    "ChangeMe-Immediately-123!"
                ),
            )
        )

    db.commit()
