import type { EventInfo, Exhibitor, Booth, RouteResult } from './types'

const API = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8000/api'

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(`${API}${path}`, {
    ...init,
    headers: { 'Content-Type': 'application/json', ...(init?.headers || {}) },
  })
  if (!response.ok) {
    const body = await response.json().catch(() => ({}))
    throw new Error(body.detail || `Request failed (${response.status})`)
  }
  return response.json() as Promise<T>
}

export const getEvent = (id: string) => request<EventInfo>(`/events/${id}`)
export const getExhibitors = (id: string, q = '') =>
  request<Exhibitor[]>(`/events/${id}/exhibitors${q ? `?q=${encodeURIComponent(q)}` : ''}`)
export const getBooths = (hallId: string) => request<Booth[]>(`/halls/${hallId}/booths`)
export const getRoute = (start_node: string, destination_node: string) =>
  request<RouteResult>('/navigation/route', {
    method: 'POST',
    body: JSON.stringify({ start_node, destination_node }),
  })

@app.post("/api/navigation/route", response_model=RouteResponse)
def calculate_route(payload: RouteRequest, db: Session = Depends(get_db)):
    nodes = {n.id for n in db.scalars(select(NavigationNode)).all()}
    if payload.start_node not in nodes or payload.destination_node not in nodes:
        raise HTTPException(status_code=404, detail="Starting point or destination was not found")
    edges = db.scalars(select(NavigationEdge).where(NavigationEdge.is_blocked == False)).all()
    graph = {}
    for edge in edges:
        graph.setdefault(edge.from_node, []).append((edge.to_node, edge.distance_m))
        graph.setdefault(edge.to_node, []).append((edge.from_node, edge.distance_m))
    queue = [(0.0, payload.start_node, [payload.start_node])]
    best = {}
    while queue:
        distance, node, path = heapq.heappop(queue)
        if node in best and best[node] <= distance: continue
        best[node] = distance
        if node == payload.destination_node:
            return RouteResponse(nodes=path, distance_m=round(distance, 2), walking_minutes=round(distance / 75, 1),
                message="Route calculated from the configured demonstration navigation graph.")
        for neighbour, weight in graph.get(node, []):
            if neighbour not in best: heapq.heappush(queue, (distance + weight, neighbour, path + [neighbour]))
    raise HTTPException(status_code=404, detail="No walkable route exists between these locations")

        raise HTTPException(status_code=400, detail="CSV requires id and name columns")

        db.add(Exhibitor(id=row["id"], event_id=event_id, name=row["name"],

            website=row.get("website") or ""))
        created += 1
    db.commit()
    return {"created": created}
